חלק א' - הבנה ותכנון

1. הבנת הפרויקט

מה אנחנו בונים?

אנחנו בונים מערכת שמאפשרת להעלות קבצי טקסט, לנתח אותם, ולקבל מידע מפורט על התוכן. המערכת תוכל לפרק טקסט למילים, למצוא דפוסים, ליצור סטטיסטיקות ועוד.

יכולות המערכת:

  • העלאת קבצים: תמיכה בקבצי .txt, .docx, .pdf
  • ניתוח טקסט: פירוק למילים, משפטים, פסקאות
  • חיפוש חכם: חיפוש מילים עם הקשר
  • ניהול קטגוריות: יצירת קבוצות מילים
  • זיהוי ביטויים: מציאת ביטויים לשוניים
  • סטטיסטיקות: נתונים על השימוש במילים

3. מודל הנתונים

הטבלאות העיקריות

בסיס הנתונים שלנו יכלול את הטבלאות הבאות:

📄 Documents

  • id (Primary Key)
  • title (כותרת המסמך)
  • file_path (נתיב הקובץ)
  • upload_date (תאריך העלאה)
  • file_size (גודל הקובץ)
  • status (סטטוס עיבוד)

📝 Sentences

  • id (Primary Key)
  • document_id (Foreign Key)
  • content (תוכן המשפט)
  • position (מיקום במסמך)
  • paragraph_number (מספר פסקה)

💬 Words

  • id (Primary Key)
  • word (המילה עצמה)
  • root_word (שורש המילה)
  • word_type (סוג המילה)
  • frequency (תדירות)

🔗 Word_Occurrences

  • id (Primary Key)
  • word_id (Foreign Key)
  • sentence_id (Foreign Key)
  • position_in_sentence (מיקום במשפט)
  • context_before (הקשר לפני)
  • context_after (הקשר אחרי)

📂 Categories

  • id (Primary Key)
  • name (שם הקטגוריה)
  • description (תיאור)
  • color (צבע לתצוגה)
  • created_date (תאריך יצירה)

🏷️ Word_Categories

  • word_id (Foreign Key)
  • category_id (Foreign Key)
  • confidence_score (רמת ביטחון)
  • assigned_by (מי הקצה)

4. יצירת בסיס הנתונים

סקריפט SQL ליצירת הטבלאות

הנה הקוד המלא ליצירת כל הטבלאות:

-- יצירת בסיס הנתונים
CREATE DATABASE text_analysis_system;

-- יצירת טבלת מסמכים
CREATE TABLE documents (
    id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    upload_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    file_size INTEGER,
    status VARCHAR(50) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- יצירת טבלת משפטים
CREATE TABLE sentences (
    id SERIAL PRIMARY KEY,
    document_id INTEGER REFERENCES documents(id) ON DELETE CASCADE,
    content TEXT NOT NULL,
    position INTEGER NOT NULL,
    paragraph_number INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- יצירת טבלת מילים
CREATE TABLE words (
    id SERIAL PRIMARY KEY,
    word VARCHAR(100) NOT NULL UNIQUE,
    root_word VARCHAR(100),
    word_type VARCHAR(50),
    frequency INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- יצירת טבלת קטגוריות
CREATE TABLE categories (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    color VARCHAR(7) DEFAULT '#007bff',
    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- יצירת טבלת הופעות מילים
CREATE TABLE word_occurrences (
    id SERIAL PRIMARY KEY,
    word_id INTEGER REFERENCES words(id) ON DELETE CASCADE,
    sentence_id INTEGER REFERENCES sentences(id) ON DELETE CASCADE,
    position_in_sentence INTEGER NOT NULL,
    context_before TEXT,
    context_after TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- יצירת טבלת קטגוריות מילים
CREATE TABLE word_categories (
    word_id INTEGER REFERENCES words(id) ON DELETE CASCADE,
    category_id INTEGER REFERENCES categories(id) ON DELETE CASCADE,
    confidence_score DECIMAL(3,2) DEFAULT 1.0,
    assigned_by VARCHAR(100) DEFAULT 'system',
    assigned_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (word_id, category_id)
);

-- יצירת אינדקסים לביצועים טובים יותר
CREATE INDEX idx_sentences_document ON sentences(document_id);
CREATE INDEX idx_word_occurrences_word ON word_occurrences(word_id);
CREATE INDEX idx_word_occurrences_sentence ON word_occurrences(sentence_id);
CREATE INDEX idx_words_frequency ON words(frequency DESC);
CREATE INDEX idx_words_word ON words(word);

def get_db_connection():

    except Exception as e:

@app.route('/api/upload', methods=['POST'])
def upload_file():
    """העלאת קובץ טקסט חדש"""
    try:
        # בדיקה שהקובץ קיים
        if 'file' not in request.files:
            return jsonify({"error": "לא נמצא קובץ בבקשה"}), 400
        
        file = request.files['file']
        if file.filename == '':
            return jsonify({"error": "לא נבחר קובץ"}), 400
        
        # שמירת הקובץ
        filename = file.filename
        file_path = os.path.join('uploads', filename)
        file.save(file_path)
        
        # שמירה בבסיס הנתונים
        conn = get_db_connection()
        if conn:
            cursor = conn.cursor()
            cursor.execute("""
                INSERT INTO documents (title, file_path, file_size, status)
                VALUES (%s, %s, %s, %s) RETURNING id
            """, (filename, file_path, os.path.getsize(file_path), 'uploaded'))
            
            document_id = cursor.fetchone()[0]
            conn.commit()
            cursor.close()
            conn.close()
            
            return jsonify({
                "message": "הקובץ הועלה בהצלחה",
                "document_id": document_id,
                "filename": filename
            })
        else:
            return jsonify({"error": "שגיאה בחיבור לבסיס הנתונים"}), 500
            
    except Exception as e:
        return jsonify({"error": f"שגיאה: {str(e)}"}), 500

@app.route('/api/analyze/', methods=['POST'])
def analyze_document(document_id):
    """ניתוח מסמך לפי ID"""
    try:
        conn = get_db_connection()
        if not conn:
            return jsonify({"error": "שגיאה בחיבור לבסיס הנתונים"}), 500
        
        # קבלת פרטי המסמך
        cursor = conn.cursor()
        cursor.execute("SELECT file_path FROM documents WHERE id = %s", (document_id,))
        result = cursor.fetchone()
        
        if not result:
            return jsonify({"error": "מסמך לא נמצא"}), 404
        
        file_path = result[0]
        
        # קריאת תוכן הקובץ
        with open(file_path, 'r', encoding='utf-8') as file:
            content = file.read()
        
        # פירוק לפסקאות ומשפטים
        paragraphs = content.split('\n\n')
        sentence_count = 0
        
        for paragraph_num, paragraph in enumerate(paragraphs, 1):
            sentences = re.split(r'[.!?]+', paragraph)
            
            for sentence in sentences:
                sentence = sentence.strip()
                if sentence:
                    sentence_count += 1
                    
                    # שמירת המשפט בבסיס הנתונים
                    cursor.execute("""
                        INSERT INTO sentences (document_id, content, position, paragraph_number)
                        VALUES (%s, %s, %s, %s) RETURNING id
                    """, (document_id, sentence, sentence_count, paragraph_num))
                    
                    sentence_id = cursor.fetchone()[0]
                    
                    # פירוק המשפט למילים
                    words = re.findall(r'\b\w+\b', sentence.lower())
                    
                    for position, word in enumerate(words):
                        # בדיקה אם המילה כבר קיימת
                        cursor.execute("SELECT id FROM words WHERE word = %s", (word,))
                        word_result = cursor.fetchone()
                        
                        if word_result:
                            word_id = word_result[0]
                            # עדכון תדירות
                            cursor.execute(
                                "UPDATE words SET frequency = frequency + 1 WHERE id = %s",
                                (word_id,)
                            )
                        else:
                            # יצירת מילה חדשה
                            cursor.execute("""
                                INSERT INTO words (word, frequency)
                                VALUES (%s, 1) RETURNING id
                            """, (word,))
                            word_id = cursor.fetchone()[0]
                        
                        # שמירת הופעת המילה
                        context_before = ' '.join(words[max(0, position-2):position])
                        context_after = ' '.join(words[position+1:min(len(words), position+3)])
                        
                        cursor.execute("""
                            INSERT INTO word_occurrences 
                            (word_id, sentence_id, position_in_sentence, context_before, context_after)
                            VALUES (%s, %s, %s, %s, %s)
                        """, (word_id, sentence_id, position, context_before, context_after))
        
        # עדכון סטטוס המסמך
        cursor.execute(
            "UPDATE documents SET status = 'analyzed' WHERE id = %s",
            (document_id,)
        )
        
        conn.commit()
        cursor.close()
        conn.close()
        
        return jsonify({
            "message": "המסמך נותח בהצלחה",
            "sentences_processed": sentence_count
        })
        
    except Exception as e:
        return jsonify({"error": f"שגיאה בניתוח: {str(e)}"}), 500

if __name__ == '__main__':
    # יצירת תיקיית uploads אם לא קיימת

    app.run(debug=True, port=5000)

import { Component } from '@angular/core';
import { HttpClient } from '@angular/common/http';

@Component({
  selector: 'app-file-upload',
  templateUrl: './file-upload.component.html',
  styleUrls: ['./file-upload.component.css']
})
export class FileUploadComponent {
  selectedFile: File | null = null;
  isLoading = false;
  
  constructor(private http: HttpClient) {}
  
  onFileSelected(event: any) {
    this.selectedFile = event.target.files[0];
  }
  
  uploadFile() {
    if (!this.selectedFile) return;
    
    this.isLoading = true;
    const formData = new FormData();
    formData.append('file', this.selectedFile);
    
    this.http.post('http://localhost:5000/api/upload', formData)
      .subscribe({
        next: (response: any) => {
          console.log('קובץ הועלה בהצלחה', response);
          this.analyzeDocument(response.document_id);
        },
        error: (error) => {
          console.error('שגיאה בהעלאה', error);
          this.isLoading = false;
        }
      });
  }
  
  analyzeDocument(documentId: number) {
    this.http.post(`http://localhost:5000/api/analyze/${documentId}`, {})
      .subscribe({
        next: (response) => {
          console.log('ניתוח הושלם', response);
          this.isLoading = false;
          // כאן תוכל להפנות לדף התוצאות
        },
        error: (error) => {
          console.error('שגיאה בניתוח', error);
          this.isLoading = false;
        }
      });
  }
}

9. חיפוש וסינון מילים

מימוש חיפוש מתקדם

נוסיף יכולת חיפוש מילים עם הצגת הקשר:

@app.route('/api/search', methods=['POST'])
def search_words():
    data = request.get_json()
    search_term = data.get('word', '').lower()
    
    if not search_term:
        return jsonify({"error": "נא להזין מילה לחיפוש"}), 400
    
    conn = get_db_connection()
    cursor = conn.cursor()
    
    # חיפוש המילה ותוצאותיה
    cursor.execute("""
        SELECT w.word, w.frequency, 
               wo.context_before, wo.context_after,
               s.content as sentence
        FROM words w
        JOIN word_occurrences wo ON w.id = wo.word_id
        JOIN sentences s ON wo.sentence_id = s.id
        WHERE w.word LIKE %s
        ORDER BY w.frequency DESC
        LIMIT 50
    """, (f'%{search_term}%',))
    
    results = cursor.fetchall()
    cursor.close()
    conn.close()
    
    return jsonify({
        "results": [
            {
                "word": result[0],
                "frequency": result[1],
                "context_before": result[2],
                "context_after": result[3],
                "sentence": result[4]
            }
            for result in results
        ]
    })

@app.route('/api/categories', methods=['GET'])
def get_categories():
    """קבלת כל הקטגוריות"""
    conn = get_db_connection()
    cursor = conn.cursor()
    
    cursor.execute("SELECT id, name, description, color FROM categories")
    categories = cursor.fetchall()
    
    cursor.close()
    conn.close()
    
    return jsonify([
        {
            "id": cat[0],
            "name": cat[1], 
            "description": cat[2],
            "color": cat[3]
        }
        for cat in categories
    ])

@app.route('/api/categories', methods=['POST'])
def create_category():
    """יצירת קטגוריה חדשה"""
    data = request.get_json()
    name = data.get('name')
    description = data.get('description', '')
    color = data.get('color', '#007bff')
    
    conn = get_db_connection()
    cursor = conn.cursor()
    
    cursor.execute("""
        INSERT INTO categories (name, description, color)
        VALUES (%s, %s, %s) RETURNING id
    """, (name, description, color))
    
    category_id = cursor.fetchone()[0]
    conn.commit()
    cursor.close()
    conn.close()
    
    return jsonify({
        "message": "קטגוריה נוצרה בהצלחה",
        "category_id": category_id
    })
